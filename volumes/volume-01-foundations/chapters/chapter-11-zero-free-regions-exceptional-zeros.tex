\chapter{المناطق الخالية من الأصفار والأصفار الاستثنائية}

\section{نطاق الفصل وحالته}

\noindent\textbf{حالة الفصل:} \textenglish{DRAFT}. يطوّر هذا الفصل عدم
الانعدام النوعي على الخط \(\Re(s)=1\)، المثبت في الفصل العاشر، إلى منطقة
كمية موحدة قرب الخط لدوال ديريشليه \(L\). ويعزل الإمكان الوحيد لصفر حقيقي
استثنائي، ثم يثبت فرادته عبر الموصلات المحدودة بمبرهنة Landau--Page، ويفصل
بين مبرهنة Siegel غير الفعالة وظاهرة Deuring--Heilbronn لتنافر بقية
الأصفار.

بدأ التأليف بعد إغلاق بوابة
\textenglish{EVIDENCE-FIRST / PRE-AUTHORING} بحكم \textenglish{PASS}. وتوجد
مواد الأدلة والتدقيق في ملفات المشروع الخاصة بالفصل الحادي عشر، وأهمها:
\begin{itemize}
  \item سجل الأدلة وخريطة البرهان في مجلد
  \texttt{research/literature-reviews}.
  \item تدقيق إشارات المشتقة اللوغاريتمية والمتراجحات الكمية.
  \item تدقيق المنطقة القياسية ومبرهنة Landau--Page.
  \item تدقيق مبرهنة Siegel وعدم الفعالية.
  \item التحقق من صيغة Deuring--Heilbronn المقتبسة.
  \item التقرير المنطقي الشامل
  \texttt{docs/CHAPTER\_11\_LOGIC\_AUDIT\_2026-07-20.md}.
\end{itemize}

لا يدعي الفصل أفضل ثابت عددي في المنطقة الخالية، ولا يثبت البرهان الكامل
لظاهرة Deuring--Heilbronn، ولا يتناول مبرهنات Siegel--Walfisz أو
Bombieri--Vinogradov أو Linnik، ولا يستعمل فرضية ريمان المعممة. وتبقى
منطقة Vinogradov--Korobov الصريحة وتقديرات كثافة الأصفار موضوعين متقدمين
مؤجلين.

\subsection{المتطلبات السابقة}

يعتمد الفصل على:
\begin{itemize}
  \item جداء هادامار للدوال التامة من الرتبة الأولى، بوصفه نتيجة قياسية في
  التحليل المركب.
  \item الشخصيات البدائية والموصل، والدالة المكتملة، والمعادلة الوظيفية،
  وتناظرات الأصفار من الفصل السابع.
  \item صيغة المشتقة اللوغاريتمية في \(\Re(s)>1\) من الفصل السابع.
  \item المتراجحة الموزونة وعدم الانعدام على الخط \(\Re(s)=1\) من الفصل
  العاشر.
  \item صيغة ستيرلينغ المنتظمة من الفصل السادس.
\end{itemize}

\subsection{نواتج التعلم}

بعد إتمام الفصل ينبغي أن يستطيع القارئ:
\begin{enumerate}
  \item تفسير سبب ظهور المقياس
  \(\log(q(|t|+2))\) في دراسة الأصفار قرب الواحد.
  \item اشتقاق صيغة حقيقية للمشتقة اللوغاريتمية بمجموع على الأصفار.
  \item ضبط عامل غاما والموصل والعوامل المحلية ضبطًا موحدًا.
  \item إثبات المنطقة القياسية الخالية مع استثناء حقيقي بسيط ممكن.
  \item التمييز بين الفرادة داخل دالة واحدة، والفرادة عند ترديد ثابت،
  وفرادة Landau--Page عبر الموصلات.
  \item إثبات مبرهنة Landau--Page من حاصل ضرب أويلري ذي معاملات غير سالبة.
  \item شرح مبرهنة Siegel ومعنى عدم فعالية ثابتها.
  \item اشتقاق حد غير فعال على بعد الصفر الاستثنائي عن الواحد.
  \item تفسير ظاهرة Deuring--Heilbronn بوصفها تنافرًا كميًا للأصفار.
\end{enumerate}

\section{المقياس التحليلي وثلاثة معاني للفرادة}

لتكن \(\chi\) شخصية بدائية غير رئيسية بموصل \(q\ge3\)، ولتكن
\[
s=\sigma+it.
\]
نستعمل في هذا الفصل المقياس
\[
\mathcal L_q(t)=\log\bigl(q(|t|+2)\bigr).
\]
يجمع هذا المقياس كلفة الموصل \(\log q\) وكلفة عامل غاما عند الارتفاع
\(|t|\). الصيغ المكافئة، مثل
\(\log(q\max\{1,|t|\})\)، لا تختلف إلا في الثوابت المطلقة؛ لكننا نثبت
تطبيعًا واحدًا حتى لا نخلط ثوابت مراجع مختلفة.

ينبغي منذ البداية فصل ثلاث عبارات:
\begin{enumerate}
  \item \textbf{داخل دالة واحدة:} قد تملك \(L(s,\chi)\) صفرًا حقيقيًا
  واحدًا فقط في المنطقة القياسية، ويكون بسيطًا.
  \item \textbf{عند ترديد ثابت:} حاصل ضرب جميع دوال الشخصيات بترديد \(q\)
  يملك على الأكثر صفرًا استثنائيًا واحدًا في منطقة صريحة مناسبة؛ انظر
  \cite{McCurley1984,Kadiri2018}.
  \item \textbf{عبر الموصلات حتى \(Q\):} مبرهنة Landau--Page تمنع وجود
  شخصيتين بدائيتين مختلفتين لهما صفران شديدا القرب من الواحد.
\end{enumerate}

هذه المستويات متقاربة في الفكرة، لكنها ليست صياغة واحدة.

\section{الرد إلى الشخصية البدائية}

\begin{proposition}[الأصفار القريبة من الواحد والجد البدائي]
\label{prop:chapter11-primitive-reduction}
\resultid{ANT-PROP-11-01}
\draftresult
لتكن \(\chi\pmod q\) شخصية مستحثة من شخصية بدائية
\(\chi^*\pmod r\). عندئذ
\[
L(s,\chi)
=
L(s,\chi^*)
\prod_{\substack{p\mid q\\p\nmid r}}
\left(1-\chi^*(p)p^{-s}\right).
\]
ولا ينعدم أي عامل محلي في \(\Re(s)>0\). لذلك تتطابق أصفار الدالتين في
هذا النصف من المستوى، مع ضبط المشتقتين اللوغاريتميتين بفارق
\(O(\log q)\) عندما \(\Re(s)>1\).
\end{proposition}

\begin{proof}
صيغة العوامل المحلية هي
\textenglish{\texttt{ANT-PROP-07-03}}. وإذا \(\sigma>0\)، فإن
\[
|\chi^*(p)p^{-s}|
\le p^{-\sigma}<1,
\]
فلا ينعدم العامل.

وبالتفاضل اللوغاريتمي:
\[
\frac{L'}{L}(s,\chi)
=
\frac{L'}{L}(s,\chi^*)
+
\sum_{\substack{p\mid q\\p\nmid r}}
\frac{\chi^*(p)(\log p)p^{-s}}
     {1-\chi^*(p)p^{-s}}.
\]
ولـ\(\sigma>1\):
\[
\left|
\frac{\chi^*(p)(\log p)p^{-s}}
     {1-\chi^*(p)p^{-s}}
\right|
\le
\frac{\log p}{p^\sigma-1}
\le\log p.
\]
ومن ثم لا يتجاوز مجموع الحدود
\[
\sum_{p\mid q}\log p
=
\log\operatorname{rad}(q)
\le\log q.
\]
\end{proof}

\section{الدالة المكتملة والمشتقة اللوغاريتمية}

ضع
\[
a_\chi=\frac{1-\chi(-1)}2\in\{0,1\},
\]
وعرف
\[
\Lambda(s,\chi)
=
\left(\frac q\pi\right)^{(s+a_\chi)/2}
\Gamma\!\left(\frac{s+a_\chi}{2}\right)L(s,\chi).
\]
الدالة \(\Lambda(s,\chi)\) تامة من الرتبة الأولى، وتحقق المعادلة
الوظيفية
\[
\Lambda(s,\chi)
=
\varepsilon_\chi
\Lambda(1-s,\overline\chi),
\qquad |\varepsilon_\chi|=1.
\]

\begin{lemma}[صيغة المشتقة اللوغاريتمية بمجموع على الأصفار]
\label{lem:chapter11-log-derivative-zeros}
\resultid{ANT-LEM-11-01}
\draftresult
إذا كانت \(\chi\) بدائية وغير رئيسية، و\(\Re(s)>1\)، فإن
\[
-\Re\frac{L'}{L}(s,\chi)
=
\frac12\log\frac q\pi
+
\frac12\Re\frac{\Gamma'}{\Gamma}
\!\left(\frac{s+a_\chi}{2}\right)
-
\sum_\rho^*
\Re\frac1{s-\rho},
\]
حيث يجري الجمع المتماثل على الأصفار غير البديهية لـ\(L(s,\chi)\).
وبخاصة، مساهمة كل صفر في \(-\Re L'/L\) سالبة.
\end{lemma}

\begin{proof}
من مبرهنة هادامار العامة للدوال التامة من الرتبة الأولى:
\[
\Lambda(s,\chi)
=
\exp(A_\chi+B_\chi s)
\prod_\rho
\left(1-\frac{s}{\rho}\right)e^{s/\rho}.
\]
بالتفاضل اللوغاريتمي، ثم بالتجميع المتماثل:
\[
\frac{\Lambda'}{\Lambda}(s,\chi)
=
b_\chi+
\sum_\rho^*\frac1{s-\rho}.
\]
تجعل المعادلة الوظيفية وتناظر الأصفار حول الخط \(\Re(s)=1/2\) الجزء
الحقيقي للثابت المنتظم \(b_\chi\) مساويًا للصفر. ومن تعريف الدالة المكتملة:
\[
\frac{\Lambda'}{\Lambda}(s,\chi)
=
\frac12\log\frac q\pi
+
\frac12\frac{\Gamma'}{\Gamma}
\!\left(\frac{s+a_\chi}{2}\right)
+
\frac{L'}{L}(s,\chi).
\]
أخذ الجزء الحقيقي وإعادة الترتيب يعطي الهوية.

وإذا \(s=\sigma+it\)، و\(\rho=\beta+i\gamma\)، فإن
\[
\Re\frac1{s-\rho}
=
\frac{\sigma-\beta}
{(\sigma-\beta)^2+(t-\gamma)^2}>0,
\]
لأن \(\sigma>1>\beta\). لذلك تظهر مساهمة الصفر بإشارة سالبة في الصيغة.
\end{proof}

\section{الحد المنتظم للموصل وغاما}

\begin{lemma}[ضبط الباقي المنتظم]
\label{lem:chapter11-uniform-remainder}
\draftresult
يوجد ثابت مطلق \(C>0\) بحيث لكل شخصية بدائية غير رئيسية \(\chi\) بموصل
\(q\)، ولكل \(1<\sigma\le2\):
\[
-\Re\frac{L'}{L}(\sigma+it,\chi)
\le
C\mathcal L_q(t).
\]
وإذا عزلنا صفرًا أو مجموعة أصفار محددة، أمكن إبقاء مساهماتها السالبة في
الطرف الأيمن، مع استمرار ضبط بقية الحدود بالحد نفسه. أما الشخصية الرئيسية
\(\chi_0\pmod q\)، فتخضع للحد
\[
-\Re\frac{L'}{L}(\sigma+it,\chi_0)
\le
\Re\frac1{\sigma-1+it}
+C\mathcal L_q(t).
\]
\end{lemma}

\begin{proof}
في الشريط \(1\le\sigma\le2\)، تعطي صيغة ستيرلينغ المنتظمة
\[
\left|
\Re\frac{\Gamma'}{\Gamma}
\!\left(\frac{\sigma+a_\chi+it}{2}\right)
\right|
\ll\log(|t|+2).
\]
عند الارتفاعات المحدودة يأتي ذلك من الاستمرارية على مجموعة مدمجة بعيدة
عن أقطاب غاما، وعند الارتفاعات الكبيرة من
\(\Gamma'/\Gamma(z)=\log z+O(1/|z|)\).

في صيغة اللمّة السابقة، يكون مجموع الأصفار مطروحًا؛ فحذفه يعطي حدًا علويًا
من رتبة
\[
\log q+\log(|t|+2)
\ll\mathcal L_q(t).
\]

أما الشخصية الرئيسية فتكتب
\[
L(s,\chi_0)
=
\zeta(s)\prod_{p\mid q}(1-p^{-s}).
\]
نحتفظ بقطب زيتا عند \(s=1\)، ونضبط عامل غاما وبقية زيتا كما سبق، بينما
لا تتجاوز كلفة العوامل المحلية \(\log q\) من البرهان السابق.
\end{proof}

\section{المتراجحة الموزونة}

نستعمل من الفصل العاشر النتيجة
\textenglish{\texttt{ANT-LEM-10-01}}:
\[
-3\frac{\zeta'}{\zeta}(\sigma)
-4\Re\frac{L'}{L}(\sigma+it,\chi)
-\Re\frac{L'}{L}(\sigma+2it,\chi^2)
\ge0
\qquad(\sigma>1).
\tag{11.1}
\]
مصدرها الهوية
\[
3+4\Re z+\Re(z^2)\ge0
\qquad(|z|\le1).
\]
والمعاملات \(3,4,1\) هي التي تجعل قطب زيتا عاجزًا عن موازنة صفر غير
حقيقي قريب جدًا من الواحد.

\section{المنطقة القياسية الخالية}

\begin{theorem}[المنطقة القياسية والاستثناء الحقيقي]
\label{thm:chapter11-standard-zero-free-region}
\resultid{ANT-THM-11-01}
\draftresult
يوجد ثابت مطلق \(c_0>0\) بحيث لكل شخصية ديريشليه بدائية غير رئيسية
\(\chi\) بموصل \(q\ge3\)، تملك \(L(s,\chi)\) على الأكثر صفرًا واحدًا في
المنطقة
\[
\Re(s)>
1-
\frac{c_0}{\log(q(|\Im(s)|+2))}.
\tag{11.2}
\]
إذا وجد الصفر، فهو حقيقي وبسيط، وتكون \(\chi\) شخصية حقيقية تربيعية.
\end{theorem}

\begin{proof}
نكتب
\[
\delta=\sigma-1>0,
\qquad
\eta=1-\beta\ge0,
\qquad
\mathcal L=\log(q(|\gamma|+2)).
\]
نختار لاحقًا
\[
\delta=\frac A{\mathcal L},
\]
حيث \(A>0\) ثابت مطلق صغير.

\paragraph{الشخصية غير الحقيقية.}
افترض أن \(\rho=\beta+i\gamma\) صفر لـ\(L(s,\chi)\)، وأن \(\chi\) غير
حقيقية. عندئذ \(\chi^2\) غير رئيسية؛ إذ لو كانت رئيسية لكانت رتبة \(\chi\)
تقسم اثنين، فتكون حقيقية.

نأخذ \(t=\gamma\) في (11.1)، ونعزل الصفر \(\rho\). من اللمّة السابقة:
\[
0
\le
\frac3\delta
-
\frac4{\delta+\eta}
+C\mathcal L.
\tag{11.3}
\]
إذا افترضنا \(\eta\le c/\mathcal L\)، وقسمنا على \(\mathcal L\)، صار
الطرف الأيمن لا يزيد على
\[
\frac3A-rac4{A+c}+C.
\]
عند \(c=0\) تساوي هذه الكمية \(-1/A+C\). نختار \(A\) صغيرًا بحيث
\(1/A>2C\)، ثم نختار \(c>0\) صغيرًا بما يكفي لاستمرار السلبية. ينتج
تناقض، فتستبعد المنطقة أصفار الشخصيات غير الحقيقية.

\paragraph{الشخصية الحقيقية والصفر غير الحقيقي.}
افترض أن \(\chi\) حقيقية، وأن \(\gamma\ne0\). عندئذ
\(\overline\rho=\beta-i\gamma\) صفر أيضًا، و\(\chi^2\) هي الشخصية الرئيسية.
لا يجوز هنا إخفاء قطب الحد الثالث في حد الخطأ.

بعد عزل الصفرين والقطب، تعطي (11.1):
\begin{align*}
0\le{}&
\frac3\delta
+
\frac{\delta}{\delta^2+4\gamma^2}
-
\frac4{\delta+\eta}
-
\frac{4(\delta+\eta)}{(\delta+\eta)^2+4\gamma^2}
+C\mathcal L.
\tag{11.4}
\end{align*}
ضع
\[
r=\frac{\delta+\eta}{\delta},
\qquad
v=\frac{2|\gamma|}{\delta}.
\]
بعد ضرب الجزء الكسري في \(\delta\)، يصبح
\[
G(r,v)
=
3+rac1{1+v^2}-\frac4r-\frac{4r}{r^2+v^2}.
\]
لدينا
\[
G(1,v)
=-1-\frac3{1+v^2}
\le-1,
\]
كما أن
\[
\left|\frac{\partial G}{\partial r}(r,v)\right|
\le8
\qquad(r\ge1,v\ge0).
\]
فإذا \(\eta\le\delta/16\)، كان \(r\le1+1/16\)، ومن مبرهنة القيمة
المتوسطة
\[
G(r,v)\le-\frac12.
\]
إذن (11.4) تعطي
\[
0\le-\frac1{2\delta}+C\mathcal L,
\]
وهو تناقض إذا اخترنا \(A\) بحيث \(1/(2A)>C\). وهكذا لا يبقى صفر غير
حقيقي لشخصية حقيقية في منطقة أصغر بثابت مطلق.

\paragraph{الصفر الحقيقي وبساطته.}
افترض الآن أن \(\chi\) حقيقية، وأن لها أصفارًا حقيقية
\(\beta_j=1-\eta_j\) في المنطقة، محسوبة مع الرتب. نأخذ \(t=0\). يجتمع
قطبا الحد الأول والحد الثالث في معامل \(4/\delta\)، وتعطي الأصفار
\[
-4\sum_j\frac1{\delta+\eta_j}.
\]
إذا كان مجموع الرتب على الأقل اثنين، وكانت \(\eta_j\le\delta/2\)، فإن
\[
-4\sum_j\frac1{\delta+\eta_j}
\le-rac{16}{3\delta}.
\]
فتصبح المتراجحة
\[
0
\le
-\frac4{3\delta}+C\log(2q),
\]
وتناقض اختيار \(\delta=A/\log(2q)\) مع \(A\) صغير. إذن يوجد على الأكثر
صفر حقيقي واحد، وتكون رتبته واحدًا.

بجمع الثوابت الصغيرة المستعملة نحصل على ثابت مطلق \(c_0\). وقد استبعدنا
الشخصيات غير الحقيقية، لذلك ترتبط الحالة الباقية بشخصية حقيقية بدائية غير
رئيسية، وهي تربيعية.
\end{proof}

\begin{remark}[الثوابت الصريحة]
المبرهنة السابقة نوعية ولا تحسن الثابت. أثبت McCurley منطقة صريحة لحاصل
ضرب دوال الشخصيات بترديد ثابت \cite{McCurley1984}، وحسن Kadiri ثوابت
عددية في نطاقات محددة \cite{Kadiri2018}. لا يجوز نقل ثابت من صيغة وتطبيقه
على تطبيع لوغاريتمي آخر من دون إعادة حساب.
\end{remark}

\begin{remark}[الشخصيات غير البدائية]
إذا كانت الشخصية غير بدائية، تنقل القضية إلى جدها البدائي بواسطة
\textenglish{\texttt{ANT-PROP-11-01}}. لا تخلق العوامل المحلية صفرًا قرب
الواحد، لكنها تضيف كلفة من رتبة \(O(\log q)\) إلى المشتقة اللوغاريتمية.
\end{remark}

\section{مبرهنة Landau--Page}

المنطقة الفردية لا تمنع، من حيث الصياغة وحدها، أن تظهر استثناءات في دوال
مختلفة. تمنع المبرهنة التالية هذا التعدد عبر الموصلات المحدودة.

\begin{theorem}[مبرهنة Landau--Page النوعية]
\label{thm:chapter11-landau-page}
\resultid{ANT-THM-11-02}
\draftresult
يوجد ثابت مطلق \(c_P>0\) بحيث لكل \(Q\ge3\)، توجد على الأكثر شخصية
ديريشليه حقيقية بدائية واحدة \(\chi\) موصلها \(q\le Q\) وتملك
\(L(s,\chi)\) صفرًا حقيقيًا \(\beta\) يحقق
\[
1-\frac{c_P}{\log Q}<\beta<1.
\]
\end{theorem}

\begin{proof}
نفترض وجود شخصيتين حقيقيتين بدائيتين متميزتين
\[
\chi_i\pmod{q_i},
\qquad q_i\le Q,
\]
ولكل منهما صفر
\[
\beta_i=1-\eta_i,
\qquad
0<\eta_i\le\frac c{\log Q}.
\]
نعرف على ترديد مشترك \(m=\operatorname{lcm}(q_1,q_2)\) الشخصية
\[
\chi(n)=\chi_1(n)\chi_2(n).
\]
هي غير رئيسية؛ فلو كانت رئيسية لاتفقت الشخصيتان البدائيتان على جميع
الأعداد الأولية إلى \(m\)، ومن ثم لكانتا الشخصية البدائية نفسها.

ضع
\[
F(s)
=
\zeta(s)L(s,\chi_1)L(s,\chi_2)L(s,\chi).
\]
في \(\sigma>1\):
\begin{align*}
-\frac{F'}{F}(\sigma)
&=
\sum_{n\ge1}
\frac{\Lambda(n)}{n^\sigma}
\bigl(1+\chi_1(n)+\chi_2(n)+\chi_1(n)\chi_2(n)\bigr)\\
&=
\sum_{n\ge1}
\frac{\Lambda(n)}{n^\sigma}
(1+\chi_1(n))(1+\chi_2(n))
\ge0.
\end{align*}
احتفظنا بشخصية حاصل الضرب كما هي، ولو كانت غير بدائية، لأن هذا يحفظ
الموجبية الدقيقة. وعند تقدير مشتقتها فقط نردها إلى جدها البدائي، الذي لا
يتجاوز موصله
\[
q_1q_2\le Q^2.
\]

ضع \(\delta=\sigma-1\). من صيغة المشتقة اللوغاريتمية، مع عزل الصفرين:
\[
0
\le
\frac1\delta
-
\frac1{\delta+\eta_1}
-
\frac1{\delta+\eta_2}
+C\log Q.
\tag{11.5}
\]
نختار
\[
\delta=\frac A{\log Q}.
\]
إذا كانت \(\eta_i\le c/\log Q\)، فإن الطرف الأيمن بعد القسمة على
\(\log Q\) لا يزيد على
\[
\frac1A-rac2{A+c}+C.
\]
عند \(c=0\) تساوي الكمية \(-1/A+C\). نختار \(A\) صغيرًا بحيث
\(1/A>2C\)، ثم \(c_P>0\) صغيرًا بما يكفي لإبقاء الكمية سالبة. هذا يناقض
(11.5)، فتنتج الفرادة.
\end{proof}

\begin{remark}[ما الذي تثبته Page؟]
لا تثبت المبرهنة وجود صفر استثنائي، ولا تنفيه. إنها تقول: إذا ظهر استثناء
شديد القرب من الواحد بين الموصلات حتى \(Q\)، فلا يظهر استثناء مستقل ثانٍ
في المجال نفسه. انظر العرض الكلاسيكي في \cite{Davenport2000} والمناقشة
الحديثة في \cite{BasakPratt2026}.
\end{remark}

\section{مبرهنة Siegel وعدم الفعالية}

\begin{theorem}[مبرهنة Siegel]
\label{thm:chapter11-siegel}
\resultid{ANT-THM-11-03}
\draftresult
لكل \(\varepsilon>0\)، يوجد ثابت \(c(\varepsilon)>0\) بحيث لكل شخصية
ديريشليه حقيقية بدائية غير رئيسية \(\chi\) بموصل \(q\):
\[
|L(1,\chi)|
\ge
c(\varepsilon)q^{-\varepsilon}.
\tag{11.6}
\]
الثابت \(c(\varepsilon)\) غير فعال.
\end{theorem}

\noindent هذه النتيجة مستشهد بها من عمل Siegel الأصلي
\cite{Siegel1935}، ويمكن مراجعة برهان تحليلي حديث في
\cite{Liu2022Siegel}. لا يعني عدم الفعالية أن الثابت غير مكتوب فحسب؛ بل
يعني أن البرهان المعروف لا يقدم خوارزمية مضمونة لحسابه من
\(\varepsilon\).

\begin{lemma}[حد المشتقة قرب الواحد]
\label{lem:chapter11-derivative-near-one}
\draftresult
يوجد ثابت مطلق \(C>0\) بحيث، إذا كانت \(\chi\) غير رئيسية بترديد \(q\)،
وكان
\[
1-\frac{c_0}{\log(2q)}\le\sigma\le1,
\]
فإن
\[
|L'(\sigma,\chi)|
\le C\log^2(2q).
\]
\end{lemma}

\begin{proof}
ضع
\[
S(x)=\sum_{n\le x}\chi(n).
\]
من الدورية وانعدام مجموع دورة كاملة:
\[
|S(x)|\le q.
\]
وبالجمع الجزئي عند \(q\):
\[
L(s,\chi)
=
\sum_{n\le q}\frac{\chi(n)}{n^s}
+
s\int_q^\infty\frac{S(x)}{x^{s+1}}\,dx
\qquad(\Re(s)>0).
\]
بالتفاضل عند \(s=\sigma\) الحقيقي:
\begin{align*}
L'(\sigma,\chi)
={}&-
\sum_{n\le q}
\frac{\chi(n)\log n}{n^\sigma}
+
\int_q^\infty\frac{S(x)}{x^{\sigma+1}}\,dx\\
&-
\sigma\int_q^\infty
\frac{S(x)\log x}{x^{\sigma+1}}\,dx.
\end{align*}
في المجال المحدد \(q^{1-\sigma}\le e^{c_0}\). ومن ثم
\[
\sum_{n\le q}\frac{\log n}{n^\sigma}
\le
q^{1-\sigma}
\sum_{n\le q}\frac{\log n}{n}
\ll\log^2(2q),
\]
كما أن التكاملين لا يتجاوزان تباعًا \(O(1)\) و\(O(\log(2q))\). تنتج
اللمّة.
\end{proof}

\begin{corollary}[الإبعاد غير الفعال للصفر الاستثنائي]
\label{cor:chapter11-siegel-zero-ineffective-distance}
\resultid{ANT-COR-11-01}
\draftresult
لكل \(\varepsilon>0\)، يوجد ثابت غير فعال
\(c_\varepsilon>0\) بحيث إذا كان \(\beta\) صفرًا استثنائيًا لدالة شخصية
حقيقية بدائية بموصل \(q\)، فإن
\[
1-\beta
\ge
c_\varepsilon q^{-\varepsilon}.
\]
\end{corollary}

\begin{proof}
لدينا
\[
L(1,\chi)
=
\int_\beta^1L'(u,\chi)\,du.
\]
فتعطي اللمّة السابقة
\[
|L(1,\chi)|
\ll
(1-\beta)\log^2(2q).
\]
نطبق مبرهنة Siegel بالأس \(\varepsilon/2\):
\[
q^{-\varepsilon/2}
\ll_\varepsilon
|L(1,\chi)|.
\]
إذن
\[
1-\beta
\gg_\varepsilon
\frac{q^{-\varepsilon/2}}{\log^2(2q)}.
\]
وبما أن \(\log^2(2q)\ll_\varepsilon q^{\varepsilon/2}\)، تنتج الصيغة بعد
امتصاص الموصلات الصغيرة في الثابت. ويرث الثابت عدم الفعالية من مبرهنة
Siegel.
\end{proof}

\begin{warning}[منع إساءة استعمال مبرهنة Siegel]
لا يجوز إدخال ثابت مبرهنة Siegel في حد خطأ ثم وصف الحد بأنه فعال. كل نتيجة
تعتمد عليه من دون بديل فعال ترث عدم الفعالية. وتوجد نتائج من نمط Tatuzawa
فعالة باستثناء شخصية واحدة محتملة، لكنها لا تزيل العقبة العامة من دون
استثناء.
\end{warning}

\section{ظاهرة Deuring--Heilbronn}

للترديد الثابت \(q\)، ضع
\[
\mathscr L_q(s)
=
\prod_{\chi\bmod q}L(s,\chi).
\]
إذا امتلكت هذه الدالة صفرًا استثنائيًا \(\beta_1\)، فإن وجوده لا يترك
بقية الأصفار في أماكنها المعتادة؛ بل يدفعها بعيدًا عن الواحد.

\begin{theorem}[ظاهرة Deuring--Heilbronn، صيغة نوعية]
\label{thm:chapter11-deuring-heilbronn}
\resultid{ANT-THM-11-04}
\draftresult
توجد ثوابت مطلقة فعالة موجبة \(c_0,c_1,c_2\) بحيث إذا كان \(q\) كبيرًا
بما يكفي، و\(T\ge2\)، وكانت \(\mathscr L_q\) تملك صفرًا استثنائيًا
\(\beta_1\) يحقق
\[
\lambda_1
=(1-\beta_1)\log(q(T+2))
\le c_0,
\]
فإن كل صفر آخر \(\rho=\beta+i\gamma\) لـ\(\mathscr L_q\)، مع
\(|\gamma|\le T\)، يحقق
\[
1-\beta
\ge
\frac{c_1}{\log(q(T+2))}
\log\!\left(\frac{c_2}{\lambda_1}\right).
\tag{11.7}
\]
\end{theorem}

\noindent النتيجة مقتبسة بصيغتها النوعية من النسخة الصريحة الحديثة
\cite{BenliGoelTwissZaman2026}. تثبت الورقة صيغة فعالة موحدة في الشريط
الحرج كله، كما تعرض صيغة غير فعالة ذات ثوابت أفضل باستعمال مبرهنة Siegel.
لا ننقل هنا الثوابت العددية، ولا ندعي البرهان الكامل، الذي يستعمل تقديرات
دون التحدب ومجاميع مولّفة وغربال Selberg.

\begin{remark}[الفرادة ليست التنافر]
مبرهنة Landau--Page تقول إن استثناءين مستقلين لا يمكن أن يقتربا معًا من
الواحد. أما Deuring--Heilbronn فتقول أكثر: إذا اقترب استثناء واحد، فإن كل
صفر آخر يُجبر على الابتعاد بمقدار يزداد لوغاريتميًا مع شدة اقتراب
\(\beta_1\).
\end{remark}

\section{الأثر في توزيع الأوليات}

تشرح الصيغة الصريحة، عند توفرها لدوال الشخصيات، سبب خطورة الصفر
الاستثنائي. بصورة إرشادية:
\[
\psi(x,\chi)
\approx
-\sum_\rho\frac{x^\rho}{\rho}
+
\text{حدود أخرى}.
\]
إذا كان \(\beta\) صفرًا حقيقيًا قريبًا من الواحد، ظهر حد كبير من رتبة
\[
-\frac{x^\beta}{\beta}.
\]
هذا الحد يفسر لماذا لا يكفي عدم الانعدام النوعي على الخط للحصول على حدود
خطأ موحدة قوية في المتتاليات الحسابية، ولماذا يجب عزل الشخصية الاستثنائية
في البراهين الكمية.

\begin{proposition}[خريطة الأثر في الصيغة الصريحة]
\label{prop:chapter11-explicit-formula-diagnostic}
\resultid{ANT-PROP-11-02}
\draftresult
المنطقة القياسية تضبط مساهمة جميع الأصفار غير الاستثنائية في الصيغة
الصريحة، بينما يساهم الصفر الحقيقي الاستثنائي، إن وجد، بحد منفرد ينبغي
إظهاره صراحة. وتمنح ظاهرة Deuring--Heilbronn تعويضًا جزئيًا: وجود ذلك الحد
الكبير يحسن مواضع بقية الأصفار.
\end{proposition}

\noindent هذه نتيجة تشخيصية في النسخة الحالية؛ أما الصيغة الصريحة الكاملة
لدوال الشخصيات، مع القطع وتقدير الأخطاء، فتبقى مؤجلة إلى فصل كمي لاحق.

\section{تمارين}

\begin{enumerate}
  \item أثبت أن
  \[
  \sum_{p\mid q}\frac{\log p}{p^\sigma-1}
  \le\log q
  \qquad(\sigma>1).
  \]

  \item تحقق من أن الشخصية غير الحقيقية لا يمكن أن يكون مربعها شخصية
  رئيسية.

  \item أثبت الحد
  \[
  \left|\frac{\partial}{\partial r}
  \left(3+\frac1{1+v^2}-\frac4r-\frac{4r}{r^2+v^2}\right)
  \right|
  \le8
  \]
  عندما \(r\ge1\) و\(v\ge0\).

  \item أكمل تفاصيل اختيار الثابتين \(A,c_0\) في برهان المنطقة القياسية،
  واستخرج قيمًا صريحة غير مثلى بدلالة ثابت الباقي \(C\).

  \item أثبت أن شخصية حاصل الضرب في برهان Landau--Page غير رئيسية إذا
  كانت \(\chi_1,\chi_2\) شخصيتين حقيقيتين بدائيتين متميزتين.

  \item تحقق مباشرة من الموجبية
  \[
  1+\chi_1(n)+\chi_2(n)+\chi_1(n)\chi_2(n)
  =(1+\chi_1(n))(1+\chi_2(n))\ge0.
  \]

  \item باستعمال دورية الشخصية فقط، أثبت
  \(|L'(\sigma,\chi)|\ll\log^2 q\) في شريط عرضه \(O(1/\log q)\) إلى
  يسار الواحد.

  \item اشرح لماذا لا تنتج مبرهنة Siegel خوارزمية لحصر جميع الأصفار
  الاستثنائية حتى موصل معلوم.

  \item قارن منطقيًا بين العبارات الثلاث: «على الأكثر صفر في دالة واحدة»،
  و«على الأكثر صفر عند ترديد ثابت»، و«على الأكثر شخصية استثنائية حتى
  موصل \(Q\)».

  \item اشرح من الصيغة الإرشادية لـ\(\psi(x,\chi)\) لماذا يصبح الحد
  \(x^\beta/\beta\) خطيرًا عندما \(\beta\) قريب من الواحد.
\end{enumerate}

\section{خلاصة الفصل}

أثبتنا أن المشتقة اللوغاريتمية للدالة المكتملة تحول هندسة الأصفار إلى
متراجحات حقيقية: كل صفر يعطي مساهمة سالبة محددة، بينما لا تتجاوز كلفة
الموصل وغاما رتبة \(\log(q(|t|+2))\). وبضم هذه البنية إلى كثيرة حدود
مثلثية غير سالبة حصلنا على المنطقة القياسية، مع إمكان استثناء واحد حقيقي
بسيط لشخصية تربيعية.

ثم أثبتنا مبرهنة Landau--Page من حاصل ضرب أويلري موجب، فصار الاستثناء
وحيدًا عبر الموصلات المحدودة. وبعد ذلك فصلنا الحد غير الفعال لمبرهنة
Siegel عن النتائج الفعالة، واشتققنا منه إبعادًا غير فعال للصفر الاستثنائي.
وأخيرًا سجلنا ظاهرة Deuring--Heilbronn: الصفر الاستثنائي، إن وجد، لا يفسد
كل شيء، بل يدفع بقية الأصفار بعيدًا.

هذه الأدوات هي المدخل الطبيعي للمرحلة الكمية التالية: مبرهنة
Siegel--Walfisz، ثم تقديرات الكثافة وBombieri--Vinogradov، مع إبقاء كل
ادعاء فعال منفصلًا عن الديون غير الفعالة.
