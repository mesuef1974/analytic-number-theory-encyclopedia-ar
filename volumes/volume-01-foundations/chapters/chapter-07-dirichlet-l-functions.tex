\chapter{دوال ديريشليه \(L\): الشخصيات والبنية التحليلية الأولى}

\section{نطاق الفصل وحالته}

\noindent\textbf{حالة الفصل:} \textenglish{DRAFT}. هذه هي الدفعة
البرهانية الأولى من الفصل السابع. سبقتها مراجعة موثقة للأهداف وخارطة
الطريق والمراجع في
\texttt{research/literature-reviews/chapter-07-dirichlet-l-functions.md}.
تثبت هذه الدفعة علاقات التعامد، وحدّ المجاميع الجزئية للشخصية غير
الرئيسية، والمنتج الأويلري، والاستمرار الهولومورفي إلى
\(\Re(s)>0\)، وعلاقة الشخصية الرئيسية بدالة زيتا. أما الموصل والشخصيات
البدائية ومجاميع غاوس والمعادلة الوظيفية وعدم الانعدام عند \(s=1\)
فتبقى للدفعات التالية.

لا يجوز الخلط بين كون \(L(1,\chi)\) قيمة منتهية للشخصية غير الرئيسية
وبين إثبات أنها غير صفرية؛ عدم الانعدام هو المدخل الحاسم لمبرهنة ديريشليه
عن الأوليات في المتتاليات الحسابية، وسيبقى خلف بوابة برهانية مستقلة.

\subsection{المتطلبات السابقة}

يفترض الفصل:

\begin{itemize}
  \item أساسيات الزمر الأبيلية المنتهية ومبرهنة بنيتها.
  \item الدوال الضربية والضربية تمامًا من الفصل الرابع.
  \item سلاسل ديريشليه وتمثيل أبيل والمنتج الأويلري من الفصل الخامس.
  \item استمرار دالة زيتا وقطبها البسيط عند \(s=1\) من الفصل السادس.
\end{itemize}

\subsection{نواتج التعلم لهذه الدفعة}

بعد إتمام هذه الدفعة ينبغي أن يستطيع القارئ:

\begin{enumerate}
  \item تعريف شخصية ديريشليه بوصفها شخصية على زمرة الوحدات ثم تمديدها
  بالصفر.
  \item إثبات الدورية والضربية التامة وتحديد القيم الممكنة للشخصية.
  \item استعمال علاقتي تعامد الشخصيات لاستخراج شروط الت合同 الحسابية.
  \item إثبات انعدام مجموع دورة كاملة للشخصية غير الرئيسية.
  \item بناء \(L(s,\chi)\) ومنتجها الأويلري عندما \(\Re(s)>1\).
  \item إثبات استمرار \(L(s,\chi)\) إلى \(\Re(s)>0\) إذا كانت
  \(\chi\) غير رئيسية.
  \item رد دالة الشخصية الرئيسية إلى زيتا مع حذف العوامل المحلية عند
  الأوليات القاسمة للترديد.
\end{enumerate}

\section{شخصيات زمرة منتهية}

لتكن \(G\) زمرة أبيلية منتهية. نسمي كل تشاكل زمر
\[
\chi:G\longrightarrow \mathbb{C}^{\times}
\]
شخصية مركبة على \(G\). ولأن \(G\) منتهية، فإن صورة \(\chi\) مؤلفة من
جذور الوحدة، ومن ثم
\[
|\chi(g)|=1,
\qquad
\overline{\chi(g)}=\chi(g)^{-1}=\chi(g^{-1}).
\]
وتشكل جميع الشخصيات زمرة تحت الضرب النقطي، تسمى الزمرة الثنائية
\(\widehat G\).

\begin{theorem}[التعامد في زمرة أبيلية منتهية]
\label{thm:finite-character-orthogonality}
\resultid{ANT-THM-07-01}
\provedhere
إذا كانت \(G\) زمرة أبيلية منتهية، فإن \(|\widehat G|=|G|\)، ولكل
\(x,y\in G\):
\[
\sum_{\chi\in\widehat G}\chi(x)\overline{\chi(y)}
=
\begin{cases}
|G|,&x=y,\\
0,&x\ne y.
\end{cases}
\]
ولكل \(\chi,\psi\in\widehat G\):
\[
\sum_{x\in G}\chi(x)\overline{\psi(x)}
=
\begin{cases}
|G|,&\chi=\psi,\\
0,&\chi\ne\psi.
\end{cases}
\]
\end{theorem}

\begin{proof}
بحسب مبرهنة بنية الزمر الأبيلية المنتهية توجد أعداد
\(m_1,\ldots,m_r\) بحيث
\[
G\cong C_{m_1}\times\cdots\times C_{m_r}.
\]
على الزمرة الدورية \(C_m=\mathbb Z/m\mathbb Z\) تكون الشخصيات بالضبط
\[
\chi_j(a)=e^{2\pi ija/m},
\qquad 0\leq j<m.
\]
ومن ثم عددها \(m\). وبأخذ حاصل الضرب على العوامل الدورية نحصل على
\(|\widehat G|=m_1\cdots m_r=|G|\)، كما تفصل الشخصيات نقاط \(G\).

نضع \(z=xy^{-1}\). إذا كان \(z=e\)، فكل حد في المجموع الأول يساوي
\(1\)، فيكون المجموع \(|G|\). وإذا كان \(z\ne e\)، فاختر شخصية
\(\eta\) تحقق \(\eta(z)\ne1\). عندئذ
\[
\eta(z)\sum_{\chi\in\widehat G}\chi(z)
=
\sum_{\chi\in\widehat G}(\eta\chi)(z)
=
\sum_{\chi\in\widehat G}\chi(z),
\]
لأن الضرب بـ\(\eta\) يبدل عناصر \(\widehat G\). لذلك المجموع صفر.
وهذا يثبت العلاقة الأولى.

أما الثانية فنطبق الحجة نفسها على الشخصية
\(\theta=\chi\overline\psi\). إذا كانت رئيسية فكل حد يساوي \(1\)،
وإلا نختار \(g\in G\) بحيث \(\theta(g)\ne1\)، ثم نضرب المجموع في
\(\theta(g)\) ونغير المتغير \(x\mapsto gx\).
\end{proof}

\begin{remark}
الجزء البنيوي المستعمل هنا هو مبرهنة تصنيف الزمر الأبيلية المنتهية. أما
علاقتا التعامد فقد اشتقتا داخل الفصل. انظر
\cite{Apostol1976,MontgomeryVaughan2007} للمسار القياسي.
\end{remark}

\section{شخصيات ديريشليه}

\begin{definition}
ليكن \(q\geq1\). شخصية ديريشليه بترديد \(q\) هي شخصية على زمرة الوحدات
\[
(\mathbb Z/q\mathbb Z)^{\times}
\]
تمدد إلى دالة على \(\mathbb Z\) بوضع
\[
\chi(n)=0
\quad\text{إذا}\quad (n,q)>1.
\]
إذا كانت الشخصية على زمرة الوحدات هي الشخصية المحايدة، سميت الشخصية
الرئيسية وكتبناها \(\chi_0\).
\end{definition}

من التعريف نحصل فورًا على:

\begin{itemize}
  \item \(\chi(n+q)=\chi(n)\) لكل \(n\in\mathbb Z\).
  \item \(\chi(mn)=\chi(m)\chi(n)\) لكل \(m,n\in\mathbb Z\).
  \item \(\chi(1)=1\)، و\(|\chi(n)|\in\{0,1\}\).
  \item إذا كان \((n,q)=1\)، فإن
  \(\overline{\chi(n)}=\chi(n)^{-1}=\chi(n^{-1})\)، حيث يؤخذ
  المعكوس بترديد \(q\).
\end{itemize}

عدد الشخصيات بترديد \(q\) يساوي
\(|(\mathbb Z/q\mathbb Z)^{\times}|=\varphi(q)\).

\begin{corollary}[تعامد شخصيات ديريشليه]
\label{cor:dirichlet-character-orthogonality}
\resultid{ANT-COR-07-01}
\provedhere
إذا كان \((ab,q)=1\)، فإن
\[
\sum_{\chi\bmod q}\chi(a)\overline{\chi(b)}
=
\begin{cases}
\varphi(q),&a\equiv b\pmod q,\\
0,&a\not\equiv b\pmod q.
\end{cases}
\]
وكذلك، لشخصيتين \(\chi,\psi\bmod q\):
\[
\sum_{a\bmod q}\chi(a)\overline{\psi(a)}
=
\begin{cases}
\varphi(q),&\chi=\psi,\\
0,&\chi\ne\psi.
\end{cases}
\]
\end{corollary}

\begin{proof}
نطبق المبرهنة السابقة على
\(G=(\mathbb Z/q\mathbb Z)^{\times}\). الحدود الموافقة للفئات غير
القابلة للعكس تساوي صفرًا تلقائيًا في المجموع الثاني.
\end{proof}

\begin{remark}[مرشح الفئة الحسابية]
إذا كان \((a,q)=1\)، فإن علاقة التعامد تعطي الهوية الدقيقة
\[
\mathbf 1_{n\equiv a\, (q)}
=
\frac1{\varphi(q)}
\sum_{\chi\bmod q}\overline{\chi(a)}\chi(n)
\]
لكل \(n\) أولي مع \(q\). هذه هي الآلية التي تحول مسألة عن فئة حسابية
إلى عائلة من المجاميع الملتوية بالشخصيات.
\end{remark}

\section{انعدام دورة كاملة وحد المجاميع الجزئية}

\begin{proposition}[المجاميع الجزئية للشخصية غير الرئيسية]
\label{prop:nonprincipal-character-partial-sums}
\resultid{ANT-PROP-07-01}
\provedhere
إذا كانت \(\chi\bmod q\) غير رئيسية، فإن
\[
\sum_{a=1}^{q}\chi(a)=0.
\]
وبالتالي، إذا
\[
S_\chi(x)=\sum_{n\leq x}\chi(n),
\]
فإن
\[
|S_\chi(x)|\leq q
\qquad(x\geq1).
\]
\end{proposition}

\begin{proof}
العلاقة الأولى هي تعامد \(\chi\) مع الشخصية الرئيسية. اكتب
\(\lfloor x\rfloor=kq+r\) حيث \(0\leq r<q\). تنعدم مساهمة كل كتلة
كاملة طولها \(q\)، فيبقى
\[
S_\chi(x)=\sum_{a=1}^{r}\chi(a).
\]
ولأن \(|\chi(a)|\leq1\)، نحصل على \(|S_\chi(x)|\leq r<q\)، ومن ثم
الحد المعلن.
\end{proof}

\begin{remark}
الحد \(q\) كاف للاستمرار التحليلي الأول، لكنه بعيد عن الحدود العميقة في
نظرية مجاميع الشخصيات، مثل متراجحتي بوليا--فينوغرادوف وبورغس. تؤجل هذه
التحسينات إلى فصل المجاميع الأسية والشخصيات.
\end{remark}

\section{تعريف دالة \(L\) والمنتج الأويلري}

\begin{definition}
إذا كانت \(\chi\bmod q\)، نعرف عندما \(\Re(s)>1\):
\[
L(s,\chi)
=
\sum_{n=1}^{\infty}\frac{\chi(n)}{n^s}.
\]
\end{definition}

بما أن \(|\chi(n)|\leq1\)، تتقارب السلسلة مطلقًا ومحليًا بانتظام في
كل نصف مستوى \(\Re(s)\geq1+\delta\)، ولذلك تعرف دالة هولومورفية في
\(\Re(s)>1\).

\begin{theorem}[المنتج الأويلري وعدم الانعدام في مجال التقارب المطلق]
\label{thm:dirichlet-l-euler-product}
\resultid{ANT-THM-07-02}
\provedhere
لكل شخصية ديريشليه \(\chi\bmod q\)، وعندما \(\Re(s)>1\):
\[
L(s,\chi)
=
\prod_p\left(1-\frac{\chi(p)}{p^s}\right)^{-1}.
\]
وفي هذا النصف من المستوى لا تنعدم \(L(s,\chi)\)، كما أن
\[
\frac1{L(s,\chi)}
=
\sum_{n=1}^{\infty}\frac{\mu(n)\chi(n)}{n^s}.
\]
\end{theorem}

\begin{proof}
الدالة \(\chi\) ضربية تمامًا، فتطبق مبرهنة المنتج الأويلري العامة من
الفصل الخامس، ويعطي العامل المحلي
\[
\sum_{k=0}^{\infty}\frac{\chi(p)^k}{p^{ks}}
=
\left(1-\chi(p)p^{-s}\right)^{-1}.
\]
أما هوية المقلوب فتنتج من
\[
(\chi)*(\mu\chi)=\varepsilon,
\]
لأن
\[
\sum_{d\mid n}\chi(d)\mu(n/d)\chi(n/d)
=
\chi(n)\sum_{d\mid n}\mu(n/d)
=
\varepsilon(n).
\]
وتتقارب سلسلة \(\mu(n)\chi(n)n^{-s}\) مطلقًا عندما \(\Re(s)>1\).
بضرب السلسلتين نحصل على \(1\)، ومن ثم لا يمكن أن تنعدم
\(L(s,\chi)\) هناك.
\end{proof}

\section{الاستمرار الأول للشخصية غير الرئيسية}

\begin{theorem}[الاستمرار إلى \(\Re(s)>0\)]
\label{thm:nonprincipal-l-continuation}
\resultid{ANT-THM-07-03}
\provedhere
إذا كانت \(\chi\bmod q\) غير رئيسية، فإن \(L(s,\chi)\) تمتد إلى دالة
هولومورفية في نصف المستوى \(\Re(s)>0\)، وفيه
\[
L(s,\chi)
=
s\int_1^{\infty}S_\chi(x)x^{-s-1}\,dx,
\qquad
S_\chi(x)=\sum_{n\leq x}\chi(n).
\]
\end{theorem}

\begin{proof}
عندما \(\Re(s)>1\)، يعطي تمثيل أبيل من الفصل الخامس
\[
\sum_{n=1}^{\infty}\frac{\chi(n)}{n^s}
=
s\int_1^{\infty}S_\chi(x)x^{-s-1}\,dx,
\]
إذ يختفي الحد الطرفي بسبب التقارب المطلق. ومن القضية
\ref{prop:nonprincipal-character-partial-sums} لدينا
\(|S_\chi(x)|\leq q\). لذلك، على مجموعة مدمجة داخل \(\Re(s)>0\)
توجد \(\delta>0\) بحيث
\[
|sS_\chi(x)x^{-s-1}|
\leq Cq\,x^{-1-\delta}.
\]
يتقارب الطرف الأيمن مطلقًا ومحليًا بانتظام، ويعرف دالة هولومورفية في
\(\Re(s)>0\). وهي توافق السلسلة في \(\Re(s)>1\)، فتكون استمرارها
التحليلي الوحيد.
\end{proof}

\begin{corollary}
إذا كانت \(\chi\) غير رئيسية، فإن
\[
L(1,\chi)
=
\int_1^{\infty}\frac{S_\chi(x)}{x^2}\,dx
\]
قيمة منتهية.
\end{corollary}

\begin{remark}[تحذير منطقي]
النتيجة السابقة تثبت الوجود والانتهاء فقط، ولا تثبت
\(L(1,\chi)\ne0\). إثبات عدم الانعدام يحتاج حجة إضافية، وهو مؤجل في هذه
النسخة حتى لا يدخل برهان مبرهنة ديريشليه في دائرة منطقية.
\end{remark}

\section{الشخصية الرئيسية وصلتها بزيتا}

\begin{proposition}[حذف العوامل المحلية]
\label{prop:principal-character-zeta-factorization}
\resultid{ANT-PROP-07-02}
\provedhere
لتكن \(\chi_0\) الشخصية الرئيسية بترديد \(q\). عندئذ، في
\(\Re(s)>1\):
\[
L(s,\chi_0)
=
\zeta(s)\prod_{p\mid q}(1-p^{-s}).
\]
وبالاستمرار الميرومورفي تصح الهوية في المستوى كله. وللدالة قطب بسيط عند
\(s=1\) وباقيه
\[
\operatorname*{Res}_{s=1}L(s,\chi_0)
=
\prod_{p\mid q}\left(1-\frac1p\right)
=
\frac{\varphi(q)}{q}.
\]
\end{proposition}

\begin{proof}
في المنتج الأويلري تكون \(\chi_0(p)=0\) إذا كان \(p\mid q\)، وتساوي
\(1\) إذا كان \(p\nmid q\). لذلك
\[
L(s,\chi_0)
=
\prod_{p\nmid q}(1-p^{-s})^{-1}
=
\zeta(s)\prod_{p\mid q}(1-p^{-s}).
\]
الجداء الأخير منته ودالة تامة، فتنتقل الهوية بالاستمرار التحليلي. وبما
أن باقي زيتا عند \(1\) يساوي \(1\)، يكون باقي الطرف الأيمن هو قيمة
الجداء المنتهي عند \(1\)، وهي \(\varphi(q)/q\).
\end{proof}

\section{أمثلة أولى}

\begin{example}[الشخصية غير الرئيسية بترديد \(4\)]
نعرف
\[
\chi_4(n)=
\begin{cases}
0,&2\mid n,\\
1,&n\equiv1\pmod4,\\
-1,&n\equiv3\pmod4.
\end{cases}
\]
فتكون
\[
L(s,\chi_4)
=
1-\frac1{3^s}+\frac1{5^s}-\frac1{7^s}+\cdots,
\]
وهي دالة بيتا لديريشليه. مجموع كل دورة كاملة يساوي صفرًا، ولذلك تمتد
هولومورفيًا إلى \(\Re(s)>0\) بالحجة السابقة.
\end{example}

\begin{example}[الشخصية الرئيسية بترديد \(4\)]
لدينا
\[
L(s,\chi_0)
=
\sum_{\substack{n\geq1\\2\nmid n}}n^{-s}
=
(1-2^{-s})\zeta(s),
\]
وباقيها عند \(s=1\) يساوي \(1/2=\varphi(4)/4\).
\end{example}

\section{أخطاء شائعة}

\begin{enumerate}
  \item \textbf{نسيان التمديد بالصفر:} شخصية ديريشليه ليست دالة بقيم على
  دائرة الوحدة لكل عدد صحيح؛ إنها تساوي صفرًا عندما لا يكون العدد أوليًا
  مع الترديد.
  \item \textbf{الخلط بين الترديد والموصل:} قد تمثل الشخصية نفسها بترديد
  غير أصغر؛ سيعالج الفصل لاحقًا الشخصية البدائية التي تستحثها.
  \item \textbf{تجاوز مجال السلسلة:} السلسلة والمنتج الأويلري معرفان
  مباشرة في \(\Re(s)>1\)، أما القيم خارج هذا النصف فتأتي من الاستمرار.
  \item \textbf{استنتاج عدم الانعدام عند \(1\) من الاستمرار:} الهولومورفية
  لا تمنع الدالة من أن تساوي صفرًا.
  \item \textbf{استعمال التعامد بلا شرط الوحدات:} صيغة مرشح الفئة تتطلب
  أن تكون الفئة أولية مع \(q\)، أو أن تراعى القيم الصفرية بدقة.
\end{enumerate}

\section{تمارين}

\begin{enumerate}
  \item اكتب جميع شخصيات ديريشليه بترديد \(3\) و\(4\)، وتحقق مباشرة من
  علاقتي التعامد.
  \item أثبت أن مرافق شخصية ديريشليه هو أيضًا شخصية، وأن
  \(\chi\overline\chi=\chi_0\) على الأعداد الأولية مع \(q\).
  \item استعمل التعامد لإثبات صيغة مرشح الفئة الحسابية.
  \item أثبت الحد الأدق
  \[
  |S_\chi(x)|\leq \min(r,q-r)
  \]
  بعد اختيار تمثيل مناسب للباقي، وبيّن موضع الحاجة إلى ترتيب حدود الدورة.
  \item اشتق
  \[
  -\frac{L'}{L}(s,\chi)
  =
  \sum_{n=1}^{\infty}\frac{\Lambda(n)\chi(n)}{n^s}
  \qquad(\Re(s)>1).
  \]
  \item احسب باقي \(L(s,\chi_0)\) عند \(1\) للترديدات
  \(q=6,8,10\).
  \item بيّن أن استمرار الشخصية غير الرئيسية إلى \(\Re(s)>0\) لا يعتمد
  على اختيار الترديد الذي تمثل به الشخصية، مع تأجيل الصياغة الكاملة إلى
  مفهوم الموصل.
\end{enumerate}

\section{حدود الدفعة التالية}

ستعالج الدفعة الثانية:

\begin{itemize}
  \item الاستحثاث والموصل وفرادة الشخصية البدائية.
  \item مجاميع غاوس وتحويل فورييه المنتهي للشخصيات.
  \item الدالة المكتملة وعامل الزوجية والمعادلة الوظيفية.
  \item الأصفار البديهية وتناظر الأصفار.
\end{itemize}

أما عدم الانعدام عند \(s=1\) ومبرهنة ديريشليه فسيبقيان دفعة مستقلة، مع
فصل حالة الشخصيات الحقيقية عن المركبة وتوثيق كل اعتماد منطقي.
