\chapter{دالة زيتا لريمان: الاستمرار التحليلي والمعادلة الوظيفية}
\personindex{ريمان، برنهارد}
\symbolindex{دالة زيتا لريمان \(\zeta(s)\)}

\section{نطاق الفصل وحالته}

\noindent\textbf{حالة الفصل:} \textenglish{REVIEWED}. اجتاز هذا الفصل
تدقيقًا علميًا داخليًا ومطابقة للمراجع القياسية وفحوص البناء، ثم اجتاز
المراجعة البشرية المستقلة. لا تعني هذه الحالة أنه
\textenglish{RELEASE-READY}؛ فالبراهين المؤجلة أدناه ما تزال ديونًا
علمية ظاهرة. ويقدم أول
بناء كامل لدالة زيتا خارج نصف مستوى سلسلتها، ويثبت المعادلة الوظيفية
والتناظرات الأساسية للأصفار. تحويل جاكوبي لدالة ثيتا مقتبس من مرجع قياسي؛
أما خطوات الاستمرار المستنتجة منه فمبرهنة هنا. ويعرض الفصل الصيغ
القياسية للنمو الرأسي وعد الأصفار ودالة هاردي، مع وسم النتائج المقتبسة.
ويعرض كذلك جداء هادامار والصيغة الصريحة والمنطقة الكلاسيكية الخالية
من الأصفار بوصفها نتائج مقتبسة؛ أما براهينها التفصيلية بمبدأ الحجة
وبيرون وتحويل المسارات فتبقى للمرحلة البرهانية التالية.

\subsection{المتطلبات السابقة}

يفترض الفصل:

\begin{itemize}
  \item الجمع الجزئي وتمثيل أبيل من الفصل الثاني.
  \item الاستمرار التحليلي ومبدأ الهوية والتقارب المحلي المنتظم من الفصل
  الثالث.
  \item سلسلة زيتا ومنتجها الأويلري في \(\Re(s)>1\) من الفصل الخامس.
  \item الخصائص الأساسية لدالة غاما وتحويل فورييه للغاوسي.
\end{itemize}

\subsection{نواتج التعلم}

بعد إتمام الفصل ينبغي أن يستطيع القارئ:

\begin{enumerate}
  \item التمييز بين دالة زيتا وتمثيلها بسلسلة ديريشليه.
  \item اشتقاق استمرارها إلى \(\Re(s)>0\) بواسطة دالة الجزء الكسري.
  \item شرح دور تحويل جاكوبي لثيتا وتمثيل ميلين.
  \item إثبات المعادلة الوظيفية للدالة المكتملة.
  \item بناء دالة ريمان \(\xi\) والتحقق من تماميتها وتناظرها.
  \item تحديد الأصفار البديهية وحصر غير البديهية في الشريط الحرج المغلق.
  \item تفسير صيغة ستيرلينغ وحد التحدب وصيغة ريمان--فون مانغولت.
  \item بناء دالة هاردي \(Z(t)\) وبيان حدود الفحص العددي.
  \item تفسير جداء هادامار القانوني والمشتقة اللوغاريتمية لـ\(\xi\).
  \item قراءة صيغة فون مانغولت الصريحة مع اصطلاح جمع الأصفار.
  \item التمييز بين منطقة خالية من الأصفار وفرضية ريمان.
  \item صياغة فرضية ريمان من دون عرضها كحقيقة مثبتة.
\end{enumerate}

\section{من السلسلة إلى الدالة}

في نصف المستوى \(\Re(s)>1\) عرّف الفصل الخامس
\[
\zeta(s)=\sum_{n=1}^{\infty}\frac1{n^s}
\]
وأثبت المنتج الأويلري
\[
\zeta(s)=\prod_p(1-p^{-s})^{-1}.
\]
هاتان الصيغتان تعطيان دالة هولومورفية غير منعدمة في ذلك النصف. لكن
السلسلة لا تتقارب عندما \(\Re(s)\leq1\)، وهذا لا يعني أن الدالة نفسها
لا يمكن استمرارها. يجب التمييز دائمًا بين مجال تمثيل ومجال الدالة
المستمرة منه.

\section{الاستمرار الأول بصيغة الجزء الكسري}

نكتب
\[
\{x\}=x-\lfloor x\rfloor,
\qquad 0\leq\{x\}<1.
\]

\begin{theorem}[استمرار زيتا إلى \(\Re(s)>0\)]
\label{thm:zeta-fractional-part-continuation}
\resultid{ANT-THM-06-01}
\provedhere
تمتد دالة زيتا ميرومورفيًا إلى نصف المستوى \(\Re(s)>0\)، وفيه
\[
\zeta(s)
=
\frac{s}{s-1}
-
s\int_1^\infty \{x\}x^{-s-1}\,dx.
\]
ولها في هذا النصف قطب وحيد بسيط عند \(s=1\)، والباقي عنده يساوي \(1\).
\end{theorem}

\begin{proof}
في \(\Re(s)>1\)، يعطي تمثيل أبيل مع
\[
A(x)=\sum_{n\leq x}1=\lfloor x\rfloor
\]
الصيغة
\[
\zeta(s)
=
s\int_1^\infty \lfloor x\rfloor x^{-s-1}\,dx.
\]
وباستعمال \(\lfloor x\rfloor=x-\{x\}\) نحصل على
\[
\zeta(s)
=
s\int_1^\infty x^{-s}\,dx
-
s\int_1^\infty \{x\}x^{-s-1}\,dx
=
\frac{s}{s-1}
-
s\int_1^\infty \{x\}x^{-s-1}\,dx.
\]
بما أن \(0\leq\{x\}<1\)، يتقارب التكامل الأخير مطلقًا ومحليًا
بانتظام عندما \(\Re(s)>0\)، ومن ثم يعرف دالة هولومورفية هناك. أما
\(s/(s-1)\) فله قطب بسيط عند \(1\) وباقيه \(1\). تتفق الصيغة الجديدة
مع السلسلة في منطقة التقاطع، فتكون استمرارها الوحيد.
\end{proof}

\begin{remark}
هذه الصيغة لا تصل وحدها إلى النصف \(\Re(s)\leq0\). للوصول إلى المستوى
كله نحتاج تناظرًا يحول السلوك قرب الصفر إلى السلوك عند اللانهاية.
\end{remark}

\section{دالة غاما والعامل الأرخميدي}
\symbolindex{دالة غاما \(\Gamma(s)\)}

نعرف، عندما \(\Re(s)>0\)،
\[
\Gamma(s)
=
\int_0^\infty e^{-x}x^{s-1}\,dx.
\]
وتحقق دالة غاما
\[
\Gamma(s+1)=s\Gamma(s),
\]
وتمتد ميرومورفيًا إلى المستوى، بأقطاب بسيطة عند الأعداد الصحيحة غير
الموجبة ومن دون أصفار. سنستعمل كذلك صيغتي الانعكاس والتضعيف
\[
\Gamma(z)\Gamma(1-z)
=
\frac{\pi}{\sin(\pi z)}
\]
و
\[
\Gamma(z)\Gamma\!\left(z+\frac12\right)
=
2^{1-2z}\sqrt{\pi}\,\Gamma(2z).
\]
هذه نتائج قياسية؛ انظر \cite{Apostol1976,Titchmarsh1986}.

\section{دالة ثيتا وتحويل جاكوبي}
\personindex{جاكوبي، كارل غوستاف}
\symbolindex{دالة ثيتا \(\Theta(t)\)}

لـ\(t>0\) نعرف
\[
\Theta(t)
=
\sum_{n\in\mathbb{Z}}e^{-\pi n^2t}
=
1+2\sum_{n=1}^{\infty}e^{-\pi n^2t}.
\]

\begin{theorem}[تحويل جاكوبي لثيتا]
\label{thm:jacobi-theta-transformation}
\resultid{ANT-THM-06-02}
\citedresult{\cite{Apostol1976,Titchmarsh1986}}
لكل \(t>0\):
\[
\Theta(t)
=
t^{-1/2}\Theta(1/t).
\]
\end{theorem}

ينتج هذا التحويل من صيغة جمع بواسون للدالة الغاوسية. والنقطة البنيوية
فيه أن قيمة ثيتا عند زمن صغير تُرد إلى قيمتها عند زمن كبير، حيث يكون
التقارب الأسي سريعًا. كما أن
\[
\Theta(t)-1=O(e^{-\pi t})
\qquad(t\to\infty).
\]

\section{تمثيل ميلين لزيتا المكتملة}
\personindex{ميلين، يلمار}
\symbolindex{دالة زيتا المكتملة \(\Lambda_\zeta(s)\)}

نعرف أولًا في \(\Re(s)>1\) الدالة المكتملة
\[
\Lambda_\zeta(s)
=
\pi^{-s/2}\Gamma\!\left(\frac{s}{2}\right)\zeta(s).
\]
بالتكامل حدًا حدًا،
\[
\int_0^\infty e^{-\pi n^2t}t^{s/2}\,\frac{dt}{t}
=
\pi^{-s/2}n^{-s}\Gamma\!\left(\frac{s}{2}\right),
\]
ومن ثم
\[
\Lambda_\zeta(s)
=
\frac12
\int_0^\infty
\bigl(\Theta(t)-1\bigr)t^{s/2}\,\frac{dt}{t}.
\]
التبديل بين المجموع والتكامل مشروع في \(\Re(s)>1\) بالتقارب المطلق.

\section{الاستمرار إلى المستوى والمعادلة الوظيفية}

\theoremindex{الاستمرار الميرومورفي والمعادلة الوظيفية لزيتا}
\begin{theorem}[الاستمرار الميرومورفي والمعادلة الوظيفية]
\label{thm:zeta-global-continuation-functional-equation}
\resultid{ANT-THM-06-03}
\provedhere
تمتد \(\zeta(s)\) ميرومورفيًا إلى \(\mathbb{C}\) كلها، وليس لها إلا
قطب بسيط عند \(s=1\) والباقي عنده يساوي \(1\). وتمتد الدالة المكتملة بحيث تحقق
\[
\Lambda_\zeta(s)=\Lambda_\zeta(1-s).
\]
وبصورة صريحة،
\[
\Lambda_\zeta(s)
=
\frac{1}{s-1}-\frac1s
+
\frac12\int_1^\infty
\bigl(\Theta(t)-1\bigr)
\left(t^{s/2}+t^{(1-s)/2}\right)
\frac{dt}{t}.
\]
\end{theorem}

\begin{proof}
نقسم تكامل ميلين عند \(t=1\). في الجزء الممتد من \(0\) إلى \(1\)
نستعمل
\[
\Theta(t)-1
=
t^{-1/2}\bigl(\Theta(1/t)-1\bigr)
+
t^{-1/2}-1,
\]
ثم نضع \(u=1/t\). ينتج
\[
\frac12\int_0^1
\bigl(\Theta(t)-1\bigr)t^{s/2}\frac{dt}{t}
=
\frac12\int_1^\infty
\bigl(\Theta(u)-1\bigr)u^{(1-s)/2}\frac{du}{u}
+
\frac{1}{s-1}-\frac1s.
\]
بإضافة الجزء من \(1\) إلى \(\infty\) نحصل على الصيغة المعلنة.

بسبب التناقص الأسي لـ\(\Theta(t)-1\)، يعرف التكامل من \(1\) إلى
\(\infty\) دالة تامة في \(s\): على كل مجموعة مدمجة تُقدَّر القوتان
بمقدار كثير حدود في \(t\)، بينما يهيمن التناقص الأسي. إذن الطرف الأيمن
ميرومورفي، وقطباه المحتملان البسيطان هما \(0\) و\(1\). وهو ثابت تحت
التبديل \(s\mapsto1-s\)، فتنتج المعادلة الوظيفية للدالة المكتملة.

وأخيرًا نكتب
\[
\zeta(s)
=
\frac{\pi^{s/2}}{\Gamma(s/2)}\Lambda_\zeta(s).
\]
الدالة \(1/\Gamma(s/2)\) تامة، وصفرها البسيط عند \(s=0\) يلغي قطب
\(\Lambda_\zeta\) هناك. أما عند \(s=1\) فلا يحدث إلغاء. وبالمطابقة
مع الاستمرار السابق يبقى الباقي عند \(1\) مساويًا \(1\).
\end{proof}

\section{دالة ريمان \(\xi\)}
\symbolindex{دالة ريمان \(\xi(s)\)}

\theoremindex{تمامية دالة ريمان \(\xi\) وتناظرها}
\begin{corollary}[تمامية \(\xi\) وتناظرها]
\label{cor:riemann-xi-entire}
\resultid{ANT-COR-06-01}
\provedhere
الدالة
\[
\xi(s)
=
\frac12 s(s-1)
\pi^{-s/2}\Gamma\!\left(\frac{s}{2}\right)\zeta(s)
\]
تامة على \(\mathbb{C}\)، وتحقق
\[
\xi(s)=\xi(1-s).
\]
\end{corollary}

\begin{proof}
يضرب العامل \(s(s-1)\) قطبي \(\Lambda_\zeta\) البسيطين عند \(0\)
و\(1\)، فتنتج دالة تامة. كما أن
\[
(1-s)(-s)=s(s-1),
\]
فتنتقل معادلة \(\Lambda_\zeta\) الوظيفية مباشرة إلى \(\xi\).
\end{proof}

\section{الصيغة الكلاسيكية للمعادلة الوظيفية}

\theoremindex{المعادلة الوظيفية الكلاسيكية لدالة زيتا}
\begin{corollary}[المعادلة الوظيفية الكلاسيكية]
\label{cor:zeta-classical-functional-equation}
\resultid{ANT-COR-06-02}
\provedhere
كهوية بين دوال ميرومورفية،
\[
\zeta(s)
=
2^s\pi^{s-1}
\sin\!\left(\frac{\pi s}{2}\right)
\Gamma(1-s)\zeta(1-s).
\]
\end{corollary}

\begin{proof}
من معادلة الدالة المكتملة نحصل أولًا على
\[
\zeta(s)
=
\pi^{s-1/2}
\frac{\Gamma((1-s)/2)}{\Gamma(s/2)}
\zeta(1-s).
\]
وبتطبيق صيغتي الانعكاس والتضعيف لغاما وتبسيط العوامل نحصل على الصيغة
المعلنة. تصح المساواة أولًا بعيدًا عن الأقطاب، ثم في كل مكان بمبدأ
الاستمرار الميرومورفي.
\end{proof}

\section{الأصفار البديهية}

\theoremindex{الأصفار البديهية لدالة زيتا}
\begin{corollary}[الأصفار البديهية]
\label{cor:zeta-trivial-zeros}
\resultid{ANT-COR-06-03}
\provedhere
لكل \(m\geq1\)، العدد
\[
s=-2m
\]
صفر بسيط لدالة زيتا.
\end{corollary}

\begin{proof}
عند \(s=-2m\) ينعدم العامل
\[
\sin\!\left(\frac{\pi s}{2}\right)
\]
انعدامًا بسيطًا، بينما تكون العوامل
\[
2^s,\quad \pi^{s-1},\quad \Gamma(1-s),\quad \zeta(1-s)
\]
منتهية وغير صفرية؛ فـ\(1-s=1+2m>1\)، وفي هذا المجال لا تنعدم زيتا
بفضل المنتج الأويلري. لذلك يكون الصفر بسيطًا.
\end{proof}

تسمى هذه الأصفار \emph{بديهية} لأنها تظهر مباشرة من عامل الجيب في
المعادلة الوظيفية، لا لأنها عديمة الأهمية.

\section{الشريط الحرج وتناظرات الأصفار}
\symbolindex{الشريط الحرج}

\begin{proposition}
\label{prop:zeta-zero-symmetries-critical-strip}
\resultid{ANT-PROP-06-01}
\provedhere
كل صفر غير بديهي لـ\(\zeta\) يقع في الشريط الحرج المغلق
\[
0\leq\Re(s)\leq1.
\]
وإذا كان \(\rho\) صفرًا غير بديهي، فإن
\[
\overline{\rho},\qquad
1-\rho,\qquad
1-\overline{\rho}
\]
أصفار أيضًا، مع الرتبة نفسها.
\end{proposition}

\begin{proof}
لا توجد أصفار في \(\Re(s)>1\) بسبب المنتج الأويلري. وإذا كان
\(\Re(s)<0\)، فإن \(\Re(1-s)>1\)، ولذلك لا ينعدم
\(\zeta(1-s)\). كما أن \(\Gamma(1-s)\) لا تنعدم ولا تملك قطبًا
هناك؛ فتبيّن المعادلة الكلاسيكية أن الأصفار الوحيدة في هذا النصف هي
أصفار عامل الجيب، أي الأعداد السالبة الزوجية.

ومن معاملات السلسلة الحقيقية في \(\Re(s)>1\) لدينا
\[
\zeta(\overline{s})=\overline{\zeta(s)},
\]
وتستمر الهوية إلى المجال الميرومورفي كله. أما التناظر حول الخط
\(\Re(s)=1/2\) فينتج من \(\xi(s)=\xi(1-s)\). وتتطابق أصفار
\(\xi\) مع الأصفار غير البديهية لزيتا: ففي الشريط الحرج المغلق، بعيدًا
عن \(0\) و\(1\)، تكون العوامل الأخرى في تعريف \(\xi\) منتهية وغير
منعدمة، بينما \(\xi(0)=\xi(1)=1/2\). وتحفظ هاتان الهويتان رتب
الأصفار.
\end{proof}

\begin{remark}
حصر الأصفار في الشريط المغلق لا يثبت خلو الخطين \(\Re(s)=0\) و
\(\Re(s)=1\). خلو الخط \(\Re(s)=1\) نتيجة عميقة تدخل في برهان مبرهنة
الأعداد الأولية، ولن تُفترض هنا.
\end{remark}

\section{النمو في الشرائط الرأسية}

لا يكفي استمرار الدالة إلى مجال أكبر؛ فتطبيقات تحويل المسارات تحتاج إلى
ضبط حجمها عندما يكبر \(|\Im(s)|\). نقطة البدء هي صيغة ستيرلينغ لعامل
غاما.

\personindex{ستيرلينغ، جيمس}
\theoremindex{صيغة ستيرلينغ المنتظمة في شريط رأسي}
\begin{theorem}[ستيرلينغ المنتظمة في شريط رأسي]
\label{thm:stirling-vertical-strip}
\resultid{ANT-THM-06-04}
\citedresult{\cite{Titchmarsh1986}}
لتكن \(a<b\) ثابتتين. بانتظام عندما
\[
a\leq\sigma\leq b,
\qquad |t|\geq2,
\qquad s=\sigma+it,
\]
لدينا، مع الفرع الرئيسي للوغاريتم،
\[
\log\Gamma(s)
=
\left(s-\frac12\right)\operatorname{Log}s-s
+\frac12\log(2\pi)
+O_{a,b}\!\left(\frac1{|t|}\right).
\]
\end{theorem}

\begin{corollary}[حجم عامل غاما الرأسي]
\label{cor:gamma-factor-vertical-size}
\resultid{ANT-COR-06-04}
\provedhere
لكل شريط ثابت \(a\leq\sigma\leq b\)، وبانتظام عندما \(|t|\geq2\):
\[
\left|
\Gamma\!\left(\frac{\sigma+it}{2}\right)
\right|
=
\sqrt{2\pi}
\left(\frac{|t|}{2}\right)^{\sigma/2-1/2}
e^{-\pi|t|/4}
\left(1+O_{a,b}\!\left(\frac1{|t|}\right)\right).
\]
\end{corollary}

\begin{proof}
نأخذ الجزء الحقيقي من صيغة ستيرلينغ بعد وضع
\(s=(\sigma+it)/2\). وبما أن
\[
\arg(\sigma+it)
=
\frac{\pi}{2}\operatorname{sgn}(t)
+O_{a,b}(1/|t|),
\]
ينتج العامل الأسي \(e^{-\pi|t|/4}\)، بينما ينتج الجزء الحقيقي من
\((s-1/2)\operatorname{Log} s\) قوة \((|t|/2)^{\sigma/2-1/2}\).
\end{proof}

يقابل التناقص الأسي لعامل غاما نموًا كثير الحدود لزيتا في المعادلة
الوظيفية. ومن الصيغ القياسية التي تجمع المعادلة الوظيفية مع مبدأ
فراجمن--ليندلوف النتيجة الآتية.

\theoremindex{حد التحدب الكلاسيكي لدالة زيتا}
\begin{theorem}[حد التحدب الكلاسيكي]
\label{thm:zeta-convexity-bound}
\resultid{ANT-THM-06-05}
\citedresult{\cite{Titchmarsh1986,IwaniecKowalski2004}}
لكل \(\varepsilon>0\)، وبانتظام عندما \(0\leq\sigma\leq1\) و
\(|t|\geq2\):
\[
\zeta(\sigma+it)
\ll_{\varepsilon}
|t|^{(1-\sigma)/2+\varepsilon}.
\]
\end{theorem}

يسمى هذا \emph{حد التحدب}، وليس أفضل حد معروف في جميع المواضع. ولا
يُستنتج منه فرض ليندلوف أو فرضية ريمان؛ بل يمثل خط أساس كلاسيكيًا تقاس
بالنسبة إليه حدود دون التحدب.

\section{عد الأصفار: صيغة ريمان--فون مانغولت}
\personindex{فون مانغولت، هانس}
\symbolindex{دالة عد الأصفار \(N(T)\)}

نرمز بـ\(N(T)\) إلى عدد الأصفار غير البديهية
\[
\rho=\beta+i\gamma,
\qquad 0<\gamma\leq T,
\]
مع حساب الرتب. وعندما يساوي \(T\) ارتفاع صفر يلزم اصطلاح طرفي؛ لذلك
نصوغ النسخة الدقيقة التالية عندما لا يكون \(T\) ارتفاع صفر.

\theoremindex{صيغة ريمان--فون مانغولت}
\begin{theorem}[صيغة ريمان--فون مانغولت]
\label{thm:riemann-von-mangoldt}
\resultid{ANT-THM-06-06}
\citedresult{\cite{Titchmarsh1986}}
إذا كان \(T\geq2\) وليس ارتفاع صفر، فإن
\[
N(T)
=
\frac{T}{2\pi}\log\!\left(\frac{T}{2\pi}\right)
-\frac{T}{2\pi}
+\frac78
+S(T)
+O\!\left(\frac1T\right),
\]
حيث
\[
S(T)
=
\frac1\pi
\arg\zeta\!\left(\frac12+iT\right),
\]
ويُحدد الجدل بالاستمرار على المسار
\[
2\longrightarrow 2+iT\longrightarrow \frac12+iT,
\]
بدءًا من \(\arg\zeta(2)=0\). وعلى وجه الخصوص،
\[
N(T)
=
\frac{T}{2\pi}\log\!\left(\frac{T}{2\pi}\right)
-\frac{T}{2\pi}
+O(\log T).
\]
\end{theorem}

الحد الرئيسي لا يحدد موضع كل صفر منفرد. لكنه يعطي كثافة متوسطة مقدارها
تقريبًا
\[
\frac{1}{2\pi}\log\!\left(\frac{T}{2\pi}\right)
\]
صفرًا لكل وحدة ارتفاع، ومن ثم يكون التباعد المتوسط قرب الارتفاع \(T\)
من رتبة
\[
\frac{2\pi}{\log(T/(2\pi))}.
\]
هذه عبارة عن المتوسط المستنتج من الحد الرئيسي، لا حد موحد لكل فجوة.

\section{دالة هاردي \(Z\)}
\personindex{هاردي، غودفري هارولد}
\symbolindex{دالة هاردي \(Z(t)\)}

نعرّف دالة ريمان--سيغل ثيتا باختيار فرع متصل بحيث
\(\vartheta_{\mathrm{RS}}(0)=0\):
\[
\vartheta_{\mathrm{RS}}(t)
=
\Im\log\Gamma\!\left(\frac14+\frac{it}{2}\right)
-\frac{t}{2}\log\pi.
\]
ثم نضع
\[
Z(t)
=
e^{i\vartheta_{\mathrm{RS}}(t)}
\zeta\!\left(\frac12+it\right).
\]

\theoremindex{واقعية دالة هاردي}
\begin{proposition}[واقعية دالة هاردي]
\label{prop:hardy-z-real}
\resultid{ANT-PROP-06-02}
\provedhere
لكل \(t\in\mathbb{R}\)، تكون \(Z(t)\) حقيقية، و
\[
Z(-t)=Z(t).
\]
كما أن
\[
Z(t)=0
\quad\Longleftrightarrow\quad
\zeta\!\left(\frac12+it\right)=0,
\]
مع حفظ رتبة الصفر.
\end{proposition}

\begin{proof}
اكتب
\[
A(t)
=
\pi^{-1/4-it/2}
\Gamma\!\left(\frac14+\frac{it}{2}\right)
=
|A(t)|e^{i\vartheta_{\mathrm{RS}}(t)}.
\]
عندئذ
\[
\Lambda_\zeta\!\left(\frac12+it\right)
=
|A(t)|Z(t).
\]
ومن المعادلة الوظيفية وهوية المرافق العقدي:
\[
\Lambda_\zeta\!\left(\frac12+it\right)
=
\Lambda_\zeta\!\left(\frac12-it\right)
=
\overline{
\Lambda_\zeta\!\left(\frac12+it\right)
},
\]
فتكون القيمة حقيقية. وبما أن \(|A(t)|>0\)، تتكافأ الأصفار وتحفظ
رتبها. أما الزوجية فتنتج من
\(\vartheta_{\mathrm{RS}}(-t)=-\vartheta_{\mathrm{RS}}(t)\)
ومن هوية المرافق لزيتا.
\end{proof}

يسمح تغير إشارة \(Z\) بعزل صفر ذي رتبة فردية على الخط الحرج. لكنه قد
يفوت صفرًا ذا رتبة زوجية، وقد تفوت شبكة خشنة تغيرين متقاربين. لذلك لا
تكفي قائمة تغيرات الإشارة لإثبات أن جميع أصفار مستطيل ما قد عُدت؛ يلزم
شهادة مستقلة من مبدأ الحجة أو من صيغة عد الأصفار.

\section{عقد التجربة العددية منخفضة الارتفاع}

تُسجل التجربة
\[
\texttt{ANT-COMP-06-01}
\]
في مجلد Python. هدفها التحقق من تنفيذ \(Z(t)\) وعزل عشرة أصفار
منخفضة الارتفاع بتغير الإشارة، ثم مطابقة الإحداثيات مع قاعدة LMFDB
الخارجية وفحص صغر الجزء التخيلي العددي، ومقارنة العدد بالحد السلس في
صيغة ريمان--فون مانغولت. سقف الدعوى صريح:

\begin{itemize}
  \item التجربة تعليمية ومنخفضة الارتفاع.
  \item لا تبحث عن أصفار خارج الخط الحرج ولا تشهد اكتمال العد بمفردها.
  \item مطابقة الإحداثيات الخارجية تتحقق من التنفيذ، ولا تستبدل شهادة

\end{itemize}

\section{جداء هادامار لدالة \(\xi\)}
\personindex{هادامار، جاك}

الاستمرار التحليلي يعطي دالة تامة، لكن تحويل أصفارها إلى جداء يحتاج
معلومة نمو عالمية. تؤدي تقديرات ثيتا وستيرلينغ إلى أن \(\xi\) دالة
تامة من الرتبة الأولى. نقتبس الصياغة القانونية التالية، بما فيها تقدير
الرتبة ومشروعية الجداء.

\theoremindex{جداء هادامار القانوني لدالة \(\xi\)}
\symbolindex{الجداء القانوني لهادامار}
\begin{theorem}[جداء هادامار القانوني]
\label{thm:xi-hadamard-product}
\resultid{ANT-THM-06-07}
\citedresult{\cite{Titchmarsh1986}}
لتكن
\[
E_1(z)=(1-z)e^z.
\]
توجد ثابتتان \(A,B\in\mathbb{C}\) بحيث
\[
\xi(s)
=
e^{A+Bs}
\prod_{\rho}
E_1\!\left(\frac{s}{\rho}\right),
\]
حيث يجري الجداء على الأصفار غير البديهية لزيتا مع حساب الرتب، ويتقارب
محليًا بانتظام بعد إدخال عوامل الجنس الأول \(E_1\). ويمكن اختيار
\[
A=\log\xi(0)=-\log2.
\]
\end{theorem}

العامل الأسي داخل \(E_1\) ليس زينة شكلية؛ فهو الذي يجعل الجداء
القانوني متقاربًا. أما الصيغة المختصرة
\[
\xi(s)=\xi(0)\prod_\rho^\ast\left(1-\frac{s}{\rho}\right)
\]
فتحتاج اصطلاح ترتيب أو اقتران متناظر للأصفار، ولهذا لا نستعملها دون
النجمة وبيان معنى الحد.

\section{المشتقة اللوغاريتمية والانتقال من الأصفار إلى الأوليات}

نكتب
\[
\psi_\Gamma(z)=\frac{\Gamma'(z)}{\Gamma(z)}
\]
لدالة ديغاما، كي لا نخلطها بدالة تشيبيشيف \(\psi(x)\).

\begin{proposition}[هويتان للمشتقة اللوغاريتمية]
\label{prop:xi-logarithmic-derivative}
\resultid{ANT-PROP-06-03}
\provedhere
بعيدًا عن الأصفار والأقطاب، ومع الاستمرار الميرومورفي حيث يلزم،
\[
\frac{\xi'}{\xi}(s)
=
B+\sum_\rho
\left(
\frac{1}{s-\rho}+\frac{1}{\rho}
\right),
\]
حيث تفهم السلسلة وفق الجداء القانوني السابق. ومن تعريف \(\xi\):
\[
\frac{\xi'}{\xi}(s)
=
\frac1s+\frac1{s-1}
-\frac12\log\pi
+\frac12\psi_\Gamma\!\left(\frac{s}{2}\right)
+\frac{\zeta'}{\zeta}(s).
\]
\end{proposition}

\begin{proof}
يعطي التفاضل اللوغاريتمي للعامل
\[
E_1(s/\rho)
=
(1-s/\rho)e^{s/\rho}
\]
المقدار
\[
\frac{1}{s-\rho}+\frac{1}{\rho}.
\]
ويجيز التقارب المحلي المنتظم للجداء القانوني التفاضل على المجموعات
المدمجة البعيدة عن الأصفار. أما الهوية الثانية فتنتج بتفاضل
\[
\xi(s)
=
\frac12s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)
\]
لوغاريتميًا حدًا حدًا.
\end{proof}

تظهر هنا الآلية البنيوية للصيغة الصريحة: المنتج الأويلري يجعل
\(-\zeta'/\zeta\) مجموعًا على القوى الأولية، بينما يجعل جداء هادامار
\(\xi'/\xi\) مجموعًا على الأصفار.

\section{صيغة فون مانغولت الصريحة}

نعرف، لكل \(x>1\)، النسخة المتماثلة عند نقاط القفز:
\[
\psi_0(x)
=
\frac12
\left(
\sum_{n<x}\Lambda(n)
+
\sum_{n\leq x}\Lambda(n)
\right).
\]
فإذا لم يكن \(x\) قوة أولية كانت \(\psi_0(x)=\psi(x)\)، وإذا كان قوة
أولية أخذت الصيغة نصف القفزة عنده.

\theoremindex{صيغة فون مانغولت الصريحة}
\begin{theorem}[الصيغة الصريحة]
\label{thm:von-mangoldt-explicit-formula}
\resultid{ANT-THM-06-08}
\citedresult{\cite{Titchmarsh1986,Davenport2000}}
لكل \(x>1\):
\[
\psi_0(x)
=
x
-
\lim_{T\to\infty}
\sum_{\substack{\rho\\|\Im\rho|\leq T}}
\frac{x^\rho}{\rho}
-
\log(2\pi)
-
\frac12\log(1-x^{-2}).
\]
تحسب الأصفار مع الرتب، ويؤخذ مجموعها بالقطع المتناظر في الارتفاع،
عبر قيم \(T\) لا تساوي ارتفاع صفر.
\end{theorem}

لا يُفهم مجموع الأصفار هنا بوصفه مجموعًا مطلقًا غير مشروط. ويأتي الحد
\(x\) من قطب زيتا عند \(1\)، ويأتي مجموع \(\rho\) من الأصفار غير
البديهية، ويجمع الحد
\[
-\frac12\log(1-x^{-2})
\]
مساهمات الأصفار البديهية \(-2,-4,\ldots\). أما الثابت
\(-\log(2\pi)\) فينتج من السلوك عند الصفر.

البرهان القياسي يبدأ بصيغة بيرون لـ\(-\zeta'/\zeta\)، ثم يحرك مسار
التكامل ويجمع البواقي. ولأن صيغة بيرون المقطوعة وشروط التحريك ما تزال
موسومة \textenglish{DEFERRED} في الفصل الثالث، نسجل الصيغة هنا كنتيجة
مقتبسة ولا ندعي اكتمال برهانها داخل الموسوعة.

\section{منطقة كلاسيكية خالية من الأصفار}

\personindex{دي لا فاليه بوسان، شارل جان}
\theoremindex{منطقة دي لا فاليه بوسان الخالية من الأصفار}
\begin{theorem}[منطقة دي لا فاليه بوسان]
\label{thm:zeta-classical-zero-free-region}
\resultid{ANT-THM-06-09}
\citedresult{\cite{Titchmarsh1986,IwaniecKowalski2004}}
يوجد ثابت مطلق \(c_0>0\) بحيث
\[
\zeta(\sigma+it)\neq0
\]
كلما
\[
|t|\geq3,
\qquad
\sigma\geq
1-\frac{c_0}{\log|t|}.
\]
\end{theorem}

\begin{remark}[علاقة هذه النتيجة بالفصل الحادي عشر]
لا تُستنتج هذه المبرهنة رسميًا من
\textenglish{\texttt{ANT-THM-11-01}} بوضع \(q=1\)، لأن تلك المبرهنة
مصاغة للشخصيات البدائية غير الرئيسية ذات الموصل \(q\ge3\).
برهان منطقة \(\zeta\) الكلاسيكية يستعمل المبدأ الموجب نفسه تقريبًا،
لكن مع معالجة مستقلة للقطب عند \(s=1\). لذلك تبقى النتيجة الحالية
\textenglish{\texttt{CITED}}، ولا ندعي أن الفصل الحادي عشر يبرهنها
حالةً خاصة.
\end{remark}

هذه المنطقة أقوى من مجرد عدم الانعدام على الخط \(\sigma=1\)، لكنها
لا تقارب دعوى فرضية ريمان؛ فهي تستبعد شريطًا رفيعًا قرب الحافة اليمنى
فقط. وتوجد مناطق حديثة أقوى من نمط فينوغرادوف--كوروبوف، لكن ثوابتها
وصيغها الصريحة تتغير مع تقدم الحدود الفعالة؛ لذلك تُحفظ في سجل الأدلة
ولا تُدمج في متن المبرهنة الكلاسيكية قبل مطابقة المصادر الأصلية.

مع صيغة بيرون وتقديرات تحويل المسار تقود المنطقة الكلاسيكية إلى صيغة
فعالة لمبرهنة الأعداد الأولية. سيظهر البرهان الكامل في فصل توزيع
الأعداد الأولية، ولن نختزل تلك السلسلة في عبارة «لا أصفار قرب \(1\)».

\section{فرضية ريمان}
\symbolindex{فرضية ريمان}

تنص فرضية ريمان على أن كل صفر غير بديهي \(\rho\) يحقق
\[
\Re(\rho)=\frac12.
\]
هذه مسألة مفتوحة، ولا توجد في الموسوعة نتيجة توسمها بالمبرهنة أو تدعي
برهانها. ستعرض الفصول اللاحقة الصيغ المكافئة والنتائج المشروطة بها مع
وسم الفرض كاملًا في كل استعمال.

\section{مثال: حساب \(\zeta(0)\)}

من الصيغة العالمية لـ\(\Lambda_\zeta\) يكون باقيها عند \(s=0\)
مساويًا \(-1\). ومن التوسع القياسي
\[
\Gamma(s/2)=\frac{2}{s}+O(1),
\qquad
\pi^{-s/2}=1+O(s),
\]
ومع هولومورفية \(\zeta\) عند الصفر نحصل على
\[
\Lambda_\zeta(s)
=
\frac{2\zeta(0)}{s}+O(1).
\]
بمقارنة الباقيين:
\[
2\zeta(0)=-1,
\qquad
\zeta(0)=-\frac12.
\]

\section{خريطة الاعتماد المنطقي}

يتبع البرهان السلسلة الآتية:

\begin{enumerate}
  \item منتج أويلر وسلسلة ديريشليه في \(\Re(s)>1\).
  \item صيغة الجزء الكسري للاستمرار إلى \(\Re(s)>0\).
  \item تحويل جاكوبي لثيتا من جمع بواسون.
  \item تمثيل ميلين للدالة المكتملة.
  \item شطر التكامل عند \(1\) واستعمال \(t\leftrightarrow1/t\).
  \item الاستمرار إلى المستوى والمعادلة الوظيفية.
  \item بناء \(\xi\) ثم استخراج تناظرات الأصفار.
  \item ضبط عامل غاما رأسيًا وفهم حد التحدب.
  \item استعمال مبدأ الحجة لاشتقاق صيغة عد الأصفار.
  \item تحويل القيم على الخط الحرج إلى دالة هاردي الحقيقية.
  \item ضبط رتبة \(\xi\) وتطبيق جداء هادامار القانوني.
  \item مساواة المشتقتين اللوغاريتميتين: مجموع على الأصفار ومجموع على
  القوى الأولية.
  \item استعمال بيرون وتحويل المسار لاستخراج الصيغة الصريحة.
  \item تطبيق تقديرات قرب \(\Re(s)=1\) لبناء منطقة خالية من الأصفار.
\end{enumerate}

فإذا حُذف تحويل ثيتا، بقي الاستمرار الأول صحيحًا، لكن لا نحصل من هذا
المسار على الاستمرار إلى المستوى كله أو التناظر \(s\leftrightarrow1-s\).

\section{أخطاء شائعة}

\begin{enumerate}
  \item اعتبار السلسلة \(\sum n^{-s}\) تعريفًا صالحًا في الشريط الحرج.
  \item استعمال المعادلة الوظيفية قبل بناء الاستمرار الميرومورفي.
  \item نسيان قطبي \(\Lambda_\zeta\) عند \(0\) و\(1\).
  \item وصف \(\Lambda_\zeta\) بأنها تامة؛ الدالة التامة هي \(\xi\).
  \item استنتاج فرضية ريمان من مجرد تناظر الأصفار.
  \item قسمة المعادلة الوظيفية على عامل يمكن أن ينعدم دون فحص الموضع.
  \item الخلط بين عدم الانعدام في \(\Re(s)>1\) وخلو الخط
  \(\Re(s)=1\).
  \item تفسير حد الخطأ \(O(\log T)\) في عد الأصفار كضبط محلي لكل فجوة.
  \item اعتبار تغيرات إشارة \(Z\) شهادة كاملة لعدم وجود أصفار فائتة.
  \item تقديم فحص عدد منته من الأصفار بوصفه دليلًا على فرضية ريمان.
  \item حذف عوامل \(e^{s/\rho}\) من جداء هادامار من دون تحديد ترتيب
  متناظر للجداء.
  \item التعامل مع مجموع الأصفار في الصيغة الصريحة كمجموع مطلق.
  \item الخلط بين منطقة خالية قرب \(\Re(s)=1\) ووقوع جميع الأصفار على
  الخط \(\Re(s)=1/2\).
\end{enumerate}

\section{أسئلة تفسير البرهان}

\begin{enumerate}
  \item لماذا يمتد تكامل الجزء الكسري إلى \(\Re(s)>0\) ولا يمتد
  مباشرة إلى \(\Re(s)>-1\)؟
  \item أين استُعمل التناقص الأسي لـ\(\Theta(t)-1\)؟
  \item لماذا يظهر القطبان \(0\) و\(1\) للدالة المكتملة، مع أن زيتا
  نفسها لا تملك قطبًا عند \(0\)؟
  \item ما العامل الذي يولد الأصفار البديهية في الصيغة الكلاسيكية؟
  \item لماذا يعطي التناظر رباعيات من الأصفار عادة، وقد تنهار الرباعية
  إلى زوج أو نقطة على خطوط التناظر؟
  \item ما الدور المختلف لكل من الحد الرئيسي و\(S(T)\) في صيغة العد؟
  \item لماذا تكون \(Z(t)\) حقيقية، ولماذا لا يكفي مسح إشاراتها لشهادة
  اكتمال الأصفار؟
  \item ما دور عامل الجنس الأول \(E_1\) في جداء هادامار؟
  \item أي حد في الصيغة الصريحة يأتي من القطب، وأيها يأتي من الأصفار
  البديهية؟
  \item لماذا لا تعني المنطقة الخالية الكلاسيكية صحة فرضية ريمان؟
\end{enumerate}

\section{تمارين}

\subsection{المستوى الأول}

\begin{enumerate}
  \item تحقق مباشرة من التقارب المحلي المنتظم للتكامل
  \[
  \int_1^\infty\{x\}x^{-s-1}\,dx
  \]
  في \(\Re(s)>0\).
  \item احسب باقي \(s/(s-1)\) عند \(s=1\).
  \item أثبت أن \(\Theta(t)-1=O(e^{-\pi t})\) عندما \(t\geq1\).
  \item تحقق من ثبات الصيغة التكاملية لـ\(\Lambda_\zeta\) تحت
  \(s\mapsto1-s\).
\end{enumerate}

\subsection{المستوى الثاني}

\begin{enumerate}
  \item أكمل تفاصيل استخراج الصيغة الكلاسيكية من معادلة الدالة المكتملة
  باستعمال هويتي غاما.
  \item أثبت أن \(\xi(0)=\xi(1)=1/2\).
  \item استخرج \(\zeta(-2m)=0\) وحدد رتبة الصفر.
  \item بيّن أن الصفر غير البديهي على الخط الحرج يقترن بمرافقه العقدي.
  \item استخرج صيغة حجم \(\Gamma((\sigma+it)/2)\) من ستيرلينغ.
  \item اشتق التباعد المتوسط من الحد الرئيسي لصيغة ريمان--فون مانغولت.
\end{enumerate}

\subsection{المستوى الثالث}

\begin{enumerate}
  \item اشتق تحويل ثيتا من صيغة جمع بواسون للغاوسي، مع ذكر اصطلاح تحويل
  فورييه المستعمل.
  \item استعمل صيغة الجزء الكسري لاستخراج تقدير لـ\(\zeta(s)\) على
  مجموعة مدمجة داخل \(\Re(s)>0\) بعيدة عن \(1\).
  \item قارن بين مسار الجزء الكسري ومسار ثيتا: ما مجال كل منهما، وما
  المعلومة الجديدة التي يعطيها الثاني؟
  \item اشرح لماذا لا تكفي المعادلة الوظيفية والمنتج الأويلري وحدهما
  لإثبات فرضية ريمان.
  \item بيّن أن تغير إشارة \(Z\) يضمن صفرًا ذا رتبة فردية، وفسّر لماذا
  لا يضمن العكس.
  \item صمم شهادة تجمع مسح \(Z\) مع قيمة مستقلة لـ\(N(T)\).
  \item اشتق هوية \(\xi'/\xi\) الثانية مباشرة من تعريف \(\xi\).
  \item تحقق من أن مجموع مساهمات الأصفار البديهية يساوي
  \[
  -\frac12\log(1-x^{-2}).
  \]
  \item اشرح بدقة أين تحتاج الصيغة الصريحة إلى قطع متناظر لمجموع
  الأصفار.
\end{enumerate}

\section{مراجع الفصل ومسار التوسع}

للبناء الكلاسيكي بواسطة ثيتا وميلين انظر
\cite{Apostol1976,Titchmarsh1986}. وللسياق التاريخي راجع
\cite{Riemann1859,Narkiewicz2000}، ولانتقال هذه البنية إلى دوال
\(L\) الأوسع انظر \cite{GelbartMiller2003,IwaniecKowalski2004}.

تعتمد تفاصيل ستيرلينغ وحد التحدب وعد الأصفار وجداء هادامار على
\cite{Titchmarsh1986,IwaniecKowalski2004}. وتعتمد الصيغة الصريحة ومسارها
إلى مبرهنة الأعداد الأولية كذلك على \cite{Davenport2000}. والمرحلة
التالية هي إتمام برهان بيرون وتحويل المسار، ثم الانتقال إلى مبرهنة
الأعداد الأولية ودوال \(L\) لديريشليه، مع إبقاء كل نتيجة مقتبسة أو
مؤجلة موسومة بحالتها.

\section{خلاصة الفصل}

بدأت زيتا كسلسلة ومنتج أويلري في نصف مستوى آمن، ثم استمرت أولًا بصيغة
الجزء الكسري، وأخيرًا إلى المستوى كله بواسطة تحويل ثيتا وتمثيل ميلين.
المعادلة الوظيفية لا تضيف تناظرًا شكليًا فقط؛ فهي التي تكشف الأصفار
البديهية وتربط نصفي المستوى حول الخط الحرج. وتضبط ستيرلينغ العامل
الأرخميدي، بينما تقيس صيغة ريمان--فون مانغولت كثافة الأصفار وتحوّلها
دالة هاردي إلى مسألة حقيقية على الخط الحرج. ويحوّل جداء هادامار الأصفار
إلى بنية ضرب، ثم تكشف الصيغة الصريحة الجسر الكمي بينها وبين القوى
الأولية. أما المنطقة الخالية الكلاسيكية فتضبط الحافة اليمنى فقط، وتبقى
فرضية ريمان دعوى مفتوحة عن موقع جميع الأصفار غير البديهية، وستظل موسومة
كذلك في كل مراحل الموسوعة.
